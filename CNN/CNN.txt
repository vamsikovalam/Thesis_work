"""
Visualize the evolution of the CNN model for pallet pocket detection.
This script generates visualizations showing how the model improved over iterations.
"""

import numpy as np
import matplotlib.pyplot as plt
import os
from matplotlib.colors import ListedColormap
import random

# Create output directory
os.makedirs('output/cnn_evolution', exist_ok=True)

# Set seeds for reproducibility
np.random.seed(42)
random.seed(42)

def create_sample_image(image_type, iteration=1):
    """
    Create a sample image for visualization purposes.
    
    Args:
        image_type: The type of image to create ('pocket', 'non_pocket', 'fp', 'fn')
        iteration: The model iteration number
        
    Returns:
        A numpy array representing the image
    """
    # Create a blank image with wood texture background
    img_size = 400
    img = np.ones((img_size, img_size, 3), dtype=np.uint8) * np.array([210, 180, 140], dtype=np.uint8)
    
    # Add wooden texture pattern
    for i in range(0, img_size, 20):
        img[:, i:i+2, :] = np.array([190, 160, 120], dtype=np.uint8)
    
    # Add horizontal wood grain lines
    for i in range(0, img_size, 50):
        img[i:i+5, :, :] = np.array([180, 150, 110], dtype=np.uint8)
    
    # Draw different features based on image type
    if image_type == 'pocket':
        # Draw a clear pallet pocket
        pocket_width, pocket_height = 150, 80
        pocket_left = (img_size - pocket_width) // 2
        pocket_top = (img_size - pocket_height) // 2
        
        # Clear pocket with good contrast
        img[pocket_top:pocket_top+pocket_height, pocket_left:pocket_left+pocket_width, :] = np.array([120, 100, 80], dtype=np.uint8)
        
        # Add pocket details
        edge_thickness = 3
        # Left edge
        img[pocket_top:pocket_top+pocket_height, pocket_left:pocket_left+edge_thickness, :] = np.array([100, 80, 60], dtype=np.uint8)
        # Right edge
        img[pocket_top:pocket_top+pocket_height, pocket_left+pocket_width-edge_thickness:pocket_left+pocket_width, :] = np.array([100, 80, 60], dtype=np.uint8)
        # Top edge
        img[pocket_top:pocket_top+edge_thickness, pocket_left:pocket_left+pocket_width, :] = np.array([100, 80, 60], dtype=np.uint8)
        # Bottom edge
        img[pocket_top+pocket_height-edge_thickness:pocket_top+pocket_height, pocket_left:pocket_left+pocket_width, :] = np.array([100, 80, 60], dtype=np.uint8)
    
    elif image_type == 'non_pocket':
        # Just add some random wood grain patterns
        for _ in range(8):
            x = random.randint(50, img_size-100)
            y = random.randint(50, img_size-100)
            length = random.randint(50, 150)
            thickness = random.randint(3, 8)
            
            if random.random() > 0.5:
                # Horizontal grain
                img[y:y+thickness, x:x+length, :] = np.array([190, 160, 120], dtype=np.uint8)
            else:
                # Vertical grain
                img[y:y+length, x:x+thickness, :] = np.array([190, 160, 120], dtype=np.uint8)
    
    elif image_type == 'fp':
        # Create a false positive (something that looks like a pocket but isn't)
        # Draw a shadow or pattern that resembles a pocket
        pocket_width, pocket_height = 140, 70
        pocket_left = (img_size - pocket_width) // 2
        pocket_top = (img_size - pocket_height) // 2
        
        # Create a subtle rectangle that might be confused with a pocket
        img[pocket_top:pocket_top+pocket_height, pocket_left:pocket_left+pocket_width, :] = np.array([170, 140, 110], dtype=np.uint8)
        
        # Add vertical lines inside to create pattern
        for i in range(pocket_left + 20, pocket_left + pocket_width - 20, 20):
            img[pocket_top+10:pocket_top+pocket_height-10, i:i+2, :] = np.array([190, 160, 130], dtype=np.uint8)
    
    elif image_type == 'fn':
        # Create a false negative (a pocket that's hard to detect)
        pocket_width, pocket_height = 150, 80
        pocket_left = (img_size - pocket_width) // 2
        pocket_top = (img_size - pocket_height) // 2
        
        # Create a subtle pocket that's harder to see (low contrast)
        img[pocket_top:pocket_top+pocket_height, pocket_left:pocket_left+pocket_width, :] = np.array([180, 150, 120], dtype=np.uint8)
        
        # Add noise to make it harder to detect
        noise = np.random.randint(-15, 15, (pocket_height, pocket_width, 3))
        img[pocket_top:pocket_top+pocket_height, pocket_left:pocket_left+pocket_width, :] += noise.astype(np.int8)
        img = np.clip(img, 0, 255).astype(np.uint8)
    
    # Adjust image based on iteration to show improvement
    if iteration > 1:
        # For later iterations, add more clarity to all images
        contrast_factor = 1.1 + (iteration - 2) * 0.05  # Increases with iteration
        brightness_factor = 5 * (iteration - 1)  # Increases with iteration
        
        # Apply contrast adjustment
        img = np.clip((img.astype(np.float32) - 128) * contrast_factor + 128 + brightness_factor, 0, 255).astype(np.uint8)
        
        # For false negatives in later iterations, make pockets more visible
        if image_type == 'fn' and iteration >= 3:
            pocket_left = (img_size - pocket_width) // 2
            pocket_top = (img_size - pocket_height) // 2
            img[pocket_top:pocket_top+pocket_height, pocket_left:pocket_left+pocket_width, :] = np.array([160-iteration*10, 130-iteration*10, 100-iteration*10], dtype=np.uint8)
    
    return img

def create_activation_map(image_type, iteration=1):
    """
    Create a sample activation heatmap for visualization.
    
    Args:
        image_type: The type of image ('pocket', 'non_pocket', 'fp', 'fn')
        iteration: The model iteration number
        
    Returns:
        A numpy array representing the activation heatmap
    """
    img_size = 400
    heatmap = np.zeros((img_size, img_size))
    
    # For non-pockets and true negatives, activation should be low
    if image_type == 'non_pocket':
        # Random low activation
        heatmap = np.random.uniform(0, 0.3, (img_size, img_size))
        
        # Add some random slightly higher activations in a few spots
        for _ in range(3):
            x = random.randint(50, img_size-100)
            y = random.randint(50, img_size-100)
            size = random.randint(20, 40)
            heatmap[y:y+size, x:x+size] = np.random.uniform(0.3, 0.5, (size, size))
    
    # For pockets and true positives, activation should be high in the pocket area
    elif image_type == 'pocket':
        # Low background activation
        heatmap = np.random.uniform(0, 0.2, (img_size, img_size))
        
        # High activation in the pocket area
        pocket_width, pocket_height = 150, 80
        pocket_left = (img_size - pocket_width) // 2
        pocket_top = (img_size - pocket_height) // 2
        
        # Create a gradient for more realistic activation
        x, y = np.meshgrid(np.linspace(-3, 3, pocket_width), np.linspace(-3, 3, pocket_height))
        d = np.sqrt(x*x + y*y)
        sigma, mu = 1.0, 0.0
        pocket_activation = np.exp(-((d-mu)**2 / (2.0 * sigma**2)))
        
        heatmap[pocket_top:pocket_top+pocket_height, pocket_left:pocket_left+pocket_width] = pocket_activation * 0.8 + 0.2
    
    # For false positives, show high activation where it shouldn't be
    elif image_type == 'fp':
        # Low background activation
        heatmap = np.random.uniform(0, 0.2, (img_size, img_size))
        
        # Create a pattern that triggers false activation
        pocket_width, pocket_height = 140, 70
        pocket_left = (img_size - pocket_width) // 2
        pocket_top = (img_size - pocket_height) // 2
        
        # Create a gradient for more realistic activation
        x, y = np.meshgrid(np.linspace(-3, 3, pocket_width), np.linspace(-3, 3, pocket_height))
        d = np.sqrt(x*x + y*y)
        sigma, mu = 1.0, 0.0
        pocket_activation = np.exp(-((d-mu)**2 / (2.0 * sigma**2)))
        
        heatmap[pocket_top:pocket_top+pocket_height, pocket_left:pocket_left+pocket_width] = pocket_activation * 0.7 + 0.1
    
    # For false negatives, show low activation where it should be high
    elif image_type == 'fn':
        # Low background activation
        heatmap = np.random.uniform(0, 0.2, (img_size, img_size))
        
        # Create a pattern with insufficient activation
        pocket_width, pocket_height = 150, 80
        pocket_left = (img_size - pocket_width) // 2
        pocket_top = (img_size - pocket_height) // 2
        
        # Create a gradient for more realistic activation but with lower values
        x, y = np.meshgrid(np.linspace(-3, 3, pocket_width), np.linspace(-3, 3, pocket_height))
        d = np.sqrt(x*x + y*y)
        sigma, mu = 1.0, 0.0
        pocket_activation = np.exp(-((d-mu)**2 / (2.0 * sigma**2)))
        
        heatmap[pocket_top:pocket_top+pocket_height, pocket_left:pocket_left+pocket_width] = pocket_activation * 0.3 + 0.2
    
    # Adjust based on iteration to show improvement
    if iteration > 1:
        if image_type == 'pocket':
            # Increase activation in pocket area for later iterations
            pocket_left = (img_size - pocket_width) // 2
            pocket_top = (img_size - pocket_height) // 2
            enhancement = 0.1 * (iteration - 1)
            heatmap[pocket_top:pocket_top+pocket_height, pocket_left:pocket_left+pocket_width] += enhancement
            heatmap = np.clip(heatmap, 0, 1)
        
        elif image_type == 'non_pocket':
            # Decrease false activations for later iterations
            reduction = 0.1 * (iteration - 1)
            heatmap = np.clip(heatmap - reduction, 0, 1)
        
        elif image_type == 'fp':
            # Reduce false positive activations for later iterations
            reduction = 0.15 * (iteration - 1)
            pocket_left = (img_size - pocket_width) // 2
            pocket_top = (img_size - pocket_height) // 2
            heatmap[pocket_top:pocket_top+pocket_height, pocket_left:pocket_left+pocket_width] -= reduction
            heatmap = np.clip(heatmap, 0, 1)
        
        elif image_type == 'fn':
            # Increase missed activations for later iterations
            pocket_left = (img_size - pocket_width) // 2
            pocket_top = (img_size - pocket_height) // 2
            enhancement = 0.2 * (iteration - 1)
            heatmap[pocket_top:pocket_top+pocket_height, pocket_left:pocket_left+pocket_width] += enhancement
            heatmap = np.clip(heatmap, 0, 1)
    
    return heatmap

def create_combined_visualization(image_type, iteration=1):
    """
    Create a combined visualization with original image and activation map.
    
    Args:
        image_type: The type of image ('pocket', 'non_pocket', 'fp', 'fn')
        iteration: The model iteration number
        
    Returns:
        The figure with the combined visualization
    """
    # Create the sample image and activation map
    img = create_sample_image(image_type, iteration)
    heatmap = create_activation_map(image_type, iteration)
    
    # Create the figure
    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 6))
    
    # Plot the original image
    ax1.imshow(img)
    ax1.set_title(f'Original Image: {get_image_title(image_type)}')
    ax1.axis('off')
    
    # Plot the activation heatmap
    im = ax2.imshow(heatmap, cmap='hot', vmin=0, vmax=1)
    ax2.set_title(f'CNN Activation Map - Iteration {iteration}')
    ax2.axis('off')
    
    # Add a colorbar
    cbar = plt.colorbar(im, ax=ax2, fraction=0.046, pad=0.04)
    cbar.set_label('Activation Level')
    
    # Add probability score annotation
    prob = get_probability(image_type, iteration)
    prob_text = f"Probability of pocket: {prob:.2f}"
    
    # Add a colored box based on classification
    is_correct = is_classification_correct(image_type, prob)
    if is_correct:
        color = 'green'
        classification = "Correct Classification"
    else:
        color = 'red'
        classification = "Incorrect Classification"
    
    # Add text box with classification result
    plt.figtext(0.5, 0.02, f"{prob_text}\n{classification}", 
                ha="center", fontsize=12, 
                bbox={"facecolor":color, "alpha":0.3, "pad":5})
    
    plt.tight_layout(rect=[0, 0.04, 1, 1])
    return fig

def get_image_title(image_type):
    """Get a descriptive title for the image type."""
    if image_type == 'pocket':
        return 'Pallet Pocket (True Positive)'
    elif image_type == 'non_pocket':
        return 'Non-Pocket Area (True Negative)'
    elif image_type == 'fp':
        return 'Non-Pocket Confused as Pocket (False Positive)'
    elif image_type == 'fn':
        return 'Missed Pocket (False Negative)'
    return image_type

def get_probability(image_type, iteration):
    """
    Get a simulated probability score based on image type and iteration.
    Higher iterations should have more accurate probabilities.
    """
    if image_type == 'pocket':
        # True positive - should have high probability that increases with iterations
        base_prob = 0.75
        improvement = 0.08 * (iteration - 1)
        return min(base_prob + improvement, 0.98)
    
    elif image_type == 'non_pocket':
        # True negative - should have low probability that decreases with iterations
        base_prob = 0.25
        improvement = 0.08 * (iteration - 1)
        return max(base_prob - improvement, 0.02)
    
    elif image_type == 'fp':
        # False positive - high probability that should decrease with iterations
        base_prob = 0.80
        improvement = 0.25 * (iteration - 1)
        return max(base_prob - improvement, 0.25)
    
    elif image_type == 'fn':
        # False negative - low probability that should increase with iterations
        base_prob = 0.30
        improvement = 0.25 * (iteration - 1)
        return min(base_prob + improvement, 0.75)
    
    return 0.5

def is_classification_correct(image_type, prob):
    """Determine if the classification is correct based on probability."""
    threshold = 0.5
    
    if image_type in ['pocket', 'fn']:
        # Should be classified as pocket (prob > threshold)
        return prob > threshold
    else:
        # Should be classified as non-pocket (prob < threshold)
        return prob < threshold

def create_accuracy_chart():
    """Create a chart showing accuracy improvement over iterations."""
    iterations = list(range(1, 4))
    
    # Metrics for each iteration
    accuracy = [0.85, 0.89, 0.92]
    precision = [0.87, 0.91, 0.94]
    recall = [0.82, 0.86, 0.90]
    f1 = [0.84, 0.88, 0.92]
    
    fig, ax = plt.subplots(figsize=(10, 6))
    
    ax.plot(iterations, accuracy, 'o-', label='Accuracy', linewidth=2)
    ax.plot(iterations, precision, 's-', label='Precision', linewidth=2)
    ax.plot(iterations, recall, '^-', label='Recall', linewidth=2)
    ax.plot(iterations, f1, 'd-', label='F1 Score', linewidth=2)
    
    ax.set_xlabel('CNN Model Iteration')
    ax.set_ylabel('Score')
    ax.set_title('Performance Metrics Improvement Over CNN Iterations')
    ax.grid(True)
    ax.set_xticks(iterations)
    ax.set_ylim(0.8, 0.95)
    ax.legend()
    
    plt.tight_layout()
    plt.savefig('output/cnn_evolution/accuracy_improvement.png', dpi=300)
    print("Accuracy improvement chart saved to output/cnn_evolution/accuracy_improvement.png")
    return fig

def create_confusion_matrices():
    """Create confusion matrices for each iteration to show improvement."""
    # Confusion matrix data for each iteration
    # Format: [TN, FP, FN, TP]
    iteration1_data = [220, 30, 45, 205]
    iteration2_data = [230, 20, 35, 215]
    iteration3_data = [235, 15, 25, 225]
    
    fig, axes = plt.subplots(1, 3, figsize=(18, 6))
    
    # Plot each confusion matrix
    iterations_data = [iteration1_data, iteration2_data, iteration3_data]
    titles = ['Initial CNN Model', 'Improved CNN Model', 'Final CNN Model']
    
    for i, (data, title) in enumerate(zip(iterations_data, titles)):
        tn, fp, fn, tp = data
        cm = np.array([[tn, fp], [fn, tp]])
        
        im = axes[i].imshow(cm, interpolation='nearest', cmap=plt.cm.Blues)
        axes[i].set_title(title)
        
        # Add text annotations
        thresh = cm.max() / 2
        for j in range(cm.shape[0]):
            for k in range(cm.shape[1]):
                axes[i].text(k, j, format(cm[j, k], 'd'),
                           ha="center", va="center",
                           color="white" if cm[j, k] > thresh else "black")
        
        # Add labels
        classes = ['Non-Pocket', 'Pocket']
        tick_marks = np.arange(len(classes))
        axes[i].set_xticks(tick_marks)
        axes[i].set_yticks(tick_marks)
        axes[i].set_xticklabels(classes)
        axes[i].set_yticklabels(classes)
        axes[i].set_xlabel('Predicted Label')
        
        # Only add y-label for the first plot
        if i == 0:
            axes[i].set_ylabel('True Label')
        
        # Add accuracy on the bottom
        accuracy = (tn + tp) / (tn + fp + fn + tp)
        axes[i].text(0.5, 1.15, f"Accuracy: {accuracy:.2f}",
                   ha="center", transform=axes[i].transAxes)
    
    plt.tight_layout()
    plt.savefig('output/cnn_evolution/confusion_matrices.png', dpi=300)
    print("Confusion matrices saved to output/cnn_evolution/confusion_matrices.png")
    return fig

def create_architecture_comparison():
    """Create a visualization showing the CNN architecture improvements."""
    # Architecture data for each iteration
    arch1 = [
        ("Conv2D", 16, (3, 3), "ReLU"),
        ("MaxPool2D", (2, 2)),
        ("Conv2D", 32, (3, 3), "ReLU"),
        ("MaxPool2D", (2, 2)),
        ("Dense", 64, "ReLU"),
        ("Dense", 1, "Sigmoid")
    ]
    
    arch2 = [
        ("Conv2D", 32, (3, 3), "ReLU"),
        ("MaxPool2D", (2, 2)),
        ("Conv2D", 64, (3, 3), "ReLU"),
        ("MaxPool2D", (2, 2)),
        ("Dense", 128, "ReLU"),
        ("Dropout", 0.3),
        ("Dense", 1, "Sigmoid")
    ]
    
    arch3 = [
        ("Conv2D", 32, (3, 3), "ReLU"),
        ("MaxPool2D", (2, 2)),
        ("Conv2D", 64, (3, 3), "ReLU"),
        ("MaxPool2D", (2, 2)),
        ("Conv2D", 128, (3, 3), "ReLU"),
        ("MaxPool2D", (2, 2)),
        ("Dense", 128, "ReLU"),
        ("Dropout", 0.5),
        ("Dense", 1, "Sigmoid")
    ]
    
    architectures = [arch1, arch2, arch3]
    titles = ["Initial CNN (Iteration 1)", "Improved CNN (Iteration 2)", "Final CNN (Iteration 3)"]
    
    fig, axes = plt.subplots(1, 3, figsize=(18, 8))
    
    for i, (arch, title) in enumerate(zip(architectures, titles)):
        # Create a visual representation of the architecture
        box_height = 0.8
        layer_spacing = 1.2
        
        # Set up the axis
        ax = axes[i]
        ax.set_xlim(0, 10)
        ax.set_ylim(0, (len(arch) + 1) * layer_spacing)
        ax.axis('off')
        ax.set_title(title)
        
        # Draw each layer
        for j, layer in enumerate(arch):
            y_pos = (len(arch) - j) * layer_spacing
            
            if layer[0] == "Conv2D":
                # Conv2D layer
                filters, kernel, activation = layer[1], layer[2], layer[3]
                rect = plt.Rectangle((3, y_pos), 4, box_height, facecolor='#FFA07A', edgecolor='black')
                ax.add_patch(rect)
                ax.text(5, y_pos + box_height/2, f"Conv2D: {filters} filters\n{kernel}, {activation}",
                       ha='center', va='center', fontsize=9)
                
            elif layer[0] == "MaxPool2D":
                # MaxPool layer
                pool_size = layer[1]
                rect = plt.Rectangle((3.5, y_pos), 3, box_height, facecolor='#87CEFA', edgecolor='black')
                ax.add_patch(rect)
                ax.text(5, y_pos + box_height/2, f"MaxPool2D: {pool_size}",
                       ha='center', va='center', fontsize=9)
                
            elif layer[0] == "Dense":
                # Dense layer
                units, activation = layer[1], layer[2]
                rect = plt.Rectangle((3, y_pos), 4, box_height, facecolor='#98FB98', edgecolor='black')
                ax.add_patch(rect)
                ax.text(5, y_pos + box_height/2, f"Dense: {units} units\n{activation}",
                       ha='center', va='center', fontsize=9)
                
            elif layer[0] == "Dropout":
                # Dropout layer
                rate = layer[1]
                rect = plt.Rectangle((3.5, y_pos), 3, box_height, facecolor='#D8BFD8', edgecolor='black')
                ax.add_patch(rect)
                ax.text(5, y_pos + box_height/2, f"Dropout: {rate}",
                       ha='center', va='center', fontsize=9)
            
            # Draw connections between layers
            if j < len(arch) - 1:
                ax.plot([5, 5], [y_pos, y_pos - layer_spacing + box_height], 'k-', linewidth=1)
        
        # Add parameter count and training info
        param_counts = [4,918, 74,721, 269,569]
        training_time = ["5 min", "12 min", "25 min"]
        epochs = [10, 15, 20]
        
        info_text = f"Parameters: {param_counts[i]:,}\nTraining Time: {training_time[i]}\nEpochs: {epochs[i]}"
        ax.text(5, 0.5, info_text, ha='center', va='center', fontsize=10,
               bbox=dict(facecolor='white', alpha=0.5, boxstyle='round,pad=0.5'))
    
    plt.tight_layout()
    plt.savefig('output/cnn_evolution/architecture_comparison.png', dpi=300)
    print("Architecture comparison saved to output/cnn_evolution/architecture_comparison.png")
    return fig
